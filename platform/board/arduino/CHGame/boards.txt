# Modified 2026-10-11: WeAct F8U6 hardware variant; see docs/weact-f8u6.md.
# CHGame board definition.
#
# Each board keeps one normal upload behaviour. The upstream CH32
# core exposes menus for clock, optimisation, debug level, C runtime and
# programmer choice; almost none of that is meaningful to someone who just
# wants to press Upload, and every extra menu is another way to end up with a
# board that will not enumerate.
#
# The advanced menus that remain are the ones a game actually needs to trade
# off: optimisation, float printf support, which core peripherals are compiled
# in, and whether USB Serial is. Normal uploads always go through the CHGame
# bootloader over USB CDC and need no programmer. The Bootloader menu and the
# two programmers (programmers.txt) are for Burn Bootloader only: changing the
# bootloader over USB, or through the factory ISP for recovery
# (docs/recovery.md).

menu.rtlib=C Runtime Library
menu.periph=Peripherals
menu.usb=USB
menu.boot=Bootloader (for Burn Bootloader)

# Relocated application: the first 12 KB of flash belong to the bootloader.
# This script and the numbers below mirror platform/bootloader/src/chgame_map.h
# in the CHGame repository. They are kept in step by hand: change them together.
rev0.build.ldscript=link_chgame_app.ld
rev0.upload.maximum_size=50944
# RAM for .data + .bss (and the heap above them): the 20,464 B after the boot
# magic, less the fixed 2 KB stack the linker script reserves at the top. The
# link already fails past this point (.stack overlaps .bss), so reporting it as
# the maximum makes the IDE's percentage the real one. The stack is separate.
rev0.upload.maximum_data_size=18416
# The IDE's "Low memory available" warning, as a percentage of the above. Its
# default of 75% fires for any sketch with a framebuffer; at 95% it means under
# ~900 B left for the heap.
rev0.build.warn_data_percentage=95

# USB flags come from the USB menu below.
rev0.build.flags.clock=-DSYSCLK_FREQ_48MHz_HSI=48000000 -DF_CPU=48000000

# The uploader performs the whole app -> bootloader -> app transition itself,
# so the IDE must NOT also touch the port: a second 1200-baud touch, or the IDE
# racing us to grab the rediscovered port, is a reliable way to break an upload
# that would otherwise work.
rev0.upload.tool=chgame_upload
rev0.upload.protocol=chgame
rev0.upload.use_1200bps_touch=false
rev0.upload.wait_for_upload_port=false
rev0.upload.native_usb=true

# Burn Bootloader writes the image chosen in the Bootloader menu below, over
# USB or through the factory ISP (Tools > Programmer, programmers.txt). The
# images are shipped with the platform so a user never has to locate or build
# a bootloader binary.
rev0.bootloader.tool=chgame_upload

rev0.vid.0=0x16C0
rev0.pid.0=0x27DD

# Link-time optimisation: the core, the libraries and the sketch are optimised
# as one program. gcc sees the LTO objects and runs the plugin itself; the one
# link flag, -flto-partition=one, keeps the program in a single partition (gcc
# splits a larger one in two and cannot inline or merge across the split,
# which cost nine of the examples 40-376 B and saved none any). Typically 1-5
# KB smaller than -Os. The default since 0.3.0: the casino games among the examples are
# written for it, and with -Os alone the larger ones do not fit. -Os links
# faster and its map file is easier to read (with LTO, functions merge and
# gain .constprop/.part names).
rev0.menu.opt.oslto=Smallest + LTO (-Os -flto, default)
rev0.menu.opt.oslto.build.flags.optimize=-Os -flto
rev0.menu.opt.oslto.build.flags.ltolink=-flto-partition=one
rev0.menu.opt.osstd=Smallest (-Os)
rev0.menu.opt.osstd.build.flags.optimize=-Os
rev0.menu.opt.o1std=Fast (-O1)
rev0.menu.opt.o1std.build.flags.optimize=-O1
rev0.menu.opt.o2std=Faster (-O2)
rev0.menu.opt.o2std.build.flags.optimize=-O2
rev0.menu.opt.o3std=Fastest (-O3)
rev0.menu.opt.o3std.build.flags.optimize=-O3
rev0.menu.opt.ogstd=Debug (-Og)
rev0.menu.opt.ogstd.build.flags.optimize=-Og

rev0.menu.rtlib.nano=Newlib Nano (default)
rev0.menu.rtlib.nano.build.flags.ldflags=--specs=nano.specs --specs=nosys.specs
rev0.menu.rtlib.nanofp=Newlib Nano + Float Printf
rev0.menu.rtlib.nanofp.build.flags.ldflags=--specs=nano.specs --specs=nosys.specs -u _printf_float

# Which core peripherals are compiled in. core.a is linked --whole-archive, so
# Serial1's constructor and the USART and timer IRQ handlers are always
# reachable and --gc-sections cannot drop them: leaving them in costs about
# 4 KB of the 50,944 B app region even in a sketch that never touches them.
# "Game" is the default because that is what the console is (USB serial, SPI
# display, buttons, a piezo driven by CHGameSound) and games need the room.
# It compiles out Serial1, tone(), analogWrite() PWM and HardwareTimer; USB
# Serial, SPI, Wire, analogRead() and attachInterrupt() are untouched.
#
# The defines live in their own property, appended to the compiler flags in
# platform.txt, rather than in compiler.*.extra_flags or build.extra_flags:
# sketches and CLI users override those, and doing so silently dropped these
# flags once already.
rev0.menu.periph.game=Game: USB Serial only, no Serial1/tone/PWM (default, ~4 KB smaller)
rev0.menu.periph.game.build.flags.periph=-DUART_MODULE_ONLY -DTIM_MODULE_ONLY
rev0.menu.periph.full=Full: Serial1 on H1, tone(), PWM, HardwareTimer
rev0.menu.periph.full.build.flags.periph=

# Serial is native USB CDC, and the 1200-baud upload handshake is handled inside
# the USB interrupt (cores/arduino/ch32/usb). Sketches need no special code.
#
# "Upload only" keeps the USB device and that handshake - both live in the
# interrupt's control path - and compiles out Serial itself (the class,
# Stream/Print and the CDC data path): about 0.6-0.7 KB. The COM port still
# appears and Upload still works with no button presses. Using Serial in that
# mode is a compile error, on purpose (see WSerial.h).
#
# There is deliberately no "None": a sketch built without USB cannot be
# re-uploaded while it runs, which breaks the promise that every sketch stays
# uploadable.
rev0.menu.usb.serial=Serial (default)
rev0.menu.usb.serial.build.usb_flags=-DUSE_CHGAME_USB_CDC
rev0.menu.usb.uploadonly=Upload only: no Serial (~0.7 KB smaller)
rev0.menu.usb.uploadonly.build.usb_flags=-DUSE_CHGAME_USB_BOOTONLY

# Which bootloader Burn Bootloader writes. It changes nothing in a sketch:
# all five keep the same memory map, USB identity and upload protocol, so a
# sketch built for one runs under the others. Two faces read the same card
# (spec/card.md): the SD Text Menu, a text list over the card's MENU.BG, and
# the SD Graphic Menu, one picture at a time and no text (the card's covers
# and the games' pictures; platform/bootloader/build.sh --ui=). Each comes in
# two styles: the selection, boxes, bar and magenta parts as one colour
# turning through the rainbow, or still (--style=: the text menu draws them
# in its text colour, the graphic menu as painted). USB Only is the same
# bootloader without the menu, for a board built without an SD card: it
# drives the LED and USB and leaves every other pin high-Z. The sources and
# what each one does are in platform/bootloader; bootloaders/CHGame/README.md
# says where each binary comes from.
rev0.menu.boot.sdmenu=SD Text Menu (Rainbow)
rev0.menu.boot.sdmenu.bootloader.file=CHGame/chgame_sdboot.bin
rev0.menu.boot.sdstatic=SD Text Menu (Static)
rev0.menu.boot.sdstatic.bootloader.file=CHGame/chgame_sdboot_static.bin
rev0.menu.boot.sdvisual=SD Graphic Menu (Rainbow)
rev0.menu.boot.sdvisual.bootloader.file=CHGame/chgame_sdvisual.bin
rev0.menu.boot.sdvisualstatic=SD Graphic Menu (Static)
rev0.menu.boot.sdvisualstatic.bootloader.file=CHGame/chgame_sdvisual_static.bin
rev0.menu.boot.nomenu=USB Only
rev0.menu.boot.nomenu.bootloader.file=CHGame/chgame_boot_nomenu.bin
