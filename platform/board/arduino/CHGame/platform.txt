# Uploader tool
# -------------
# chgame-upload owns the ENTIRE transition: it finds the application port, does
# the 1200-baud touch, waits for the bootloader to enumerate, flashes, verifies
# and launches the sketch again. That is why boards.txt disables the IDE's own
# 1200bps touch and port-waiting -- two actors racing for the same port is the
# classic source of flaky native-USB uploads.
#
# chgame-upload is a self-contained binary delivered as a tool dependency,
# the same way avrdude, bossac and openocd are. Arduino ships no Python
# interpreter, so a script here would mean either requiring a Python install
# or freezing one per platform - both worse than a 2 MB static binary.
# Its source is platform/bootloader/host/go in the CHGame repository.
tools.chgame_upload.path={runtime.tools.chgame-upload.path}
tools.chgame_upload.cmd=chgame-upload
tools.chgame_upload.cmd.windows=chgame-upload.exe

# wchisp arrives as a tool dependency, so its location is known rather than
# searched for. A board package that installs cleanly and then cannot find its
# own programmer the first time you use it is exactly the failure worth
# designing out.
tools.chgame_upload.wchisp={runtime.tools.wchisp.path}/wchisp
tools.chgame_upload.wchisp.windows={runtime.tools.wchisp.path}/wchisp.exe
tools.chgame_upload.upload.params.verbose=-verbose
tools.chgame_upload.upload.params.quiet=-quiet
tools.chgame_upload.upload.pattern={build.chgame_host} {upload.verbose} -port "{serial.port}" flash "{build.path}/{build.project_name}.bin" -device {build.chgame_device} -run

# Arduino runs a separate chip-erase step before Burn Bootloader. Both ways of
# writing a bootloader erase what they need themselves, so this is an explicit
# no-op rather than a second erase.
tools.chgame_upload.erase.params.verbose=-verbose
tools.chgame_upload.erase.params.quiet=-quiet
tools.chgame_upload.erase.pattern={build.chgame_host} noop

# Burn Bootloader writes the bootloader chosen in Tools > Bootloader
# ({bootloader.file}, boards.txt). The programmer chosen in Tools > Programmer
# sets {burn.method} (programmers.txt):
#   usb  through the bootloader already on the board, like an upload: no
#        driver, no buttons. The port is the selected one, or the board is
#        looked for if none is selected.
#   isp  through the chip's factory ISP. The board must already be in ISP mode
#        (hold BOOT across a power cycle).
# Either way no sketch is left installed: the board comes up in the bootloader
# and the next sketch goes on with a normal Upload (or from the SD card menu).
# {burn.port} is the port argument, and only the usb programmer gives one: a
# board in ISP mode has no serial port, and Arduino refuses to run a recipe
# that names {serial.port} when none is selected.
tools.chgame_upload.burn.method=usb
tools.chgame_upload.burn.port=-port "{serial.port}"
tools.chgame_upload.bootloader.params.verbose=-verbose
tools.chgame_upload.bootloader.params.quiet=-quiet
tools.chgame_upload.bootloader.pattern={build.chgame_host} {bootloader.verbose} {burn.port} burn -method {burn.method} -wchisp "{tools.chgame_upload.wchisp}" -bootloader "{runtime.platform.path}/bootloaders/{bootloader.file}"

# Upload Using Programmer: the same, then this sketch. With the factory ISP it
# is one write, and puts a board with no working bootloader straight back to
# running the sketch in front of you.
tools.chgame_upload.program.params.verbose=-verbose
tools.chgame_upload.program.params.quiet=-quiet
tools.chgame_upload.program.pattern={build.chgame_host} {program.verbose} {burn.port} burn -method {burn.method} -wchisp "{tools.chgame_upload.wchisp}" -bootloader "{runtime.platform.path}/bootloaders/{bootloader.file}" -app "{build.path}/{build.project_name}.bin"
